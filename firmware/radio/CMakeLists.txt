cmake_minimum_required(VERSION 3.20)
project(smart_blinds_radio_window LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(radio_window_tests test_radio_window.cpp)
target_include_directories(radio_window_tests PRIVATE ../include)
target_compile_options(radio_window_tests PRIVATE -Wall -Wextra -Werror -pedantic)
enable_testing()
add_test(NAME radio_window COMMAND radio_window_tests)
