cmake_minimum_required(VERSION 3.20)
project(smart_blinds_control LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(blinds_control src/control.cpp)
target_include_directories(blinds_control PUBLIC include)
target_compile_options(blinds_control PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(control_tests tests/control_tests.cpp)
target_link_libraries(control_tests PRIVATE blinds_control)
target_compile_options(control_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(group_demo examples/group_demo.cpp)
target_link_libraries(group_demo PRIVATE blinds_control)
enable_testing()
add_test(NAME control_tests COMMAND control_tests)
add_executable(heartbeat_tests tests/heartbeat_tests.cpp)
target_include_directories(heartbeat_tests PRIVATE include)
target_compile_options(heartbeat_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME heartbeat_tests COMMAND heartbeat_tests)

add_executable(clock_tests tests/clock_tests.cpp)
target_include_directories(clock_tests PRIVATE include)
target_compile_options(clock_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME clock_tests COMMAND clock_tests)
add_executable(charger_tests tests/charger_tests.cpp src/charger.cpp)
target_include_directories(charger_tests PRIVATE include)
target_compile_options(charger_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME charger_tests COMMAND charger_tests)
add_executable(board_io_tests tests/board_io_tests.cpp)
target_include_directories(board_io_tests PRIVATE include)
target_compile_options(board_io_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME board_io_tests COMMAND board_io_tests)
add_library(blinds_motion src/motion_executor.cpp)
target_link_libraries(blinds_motion PUBLIC blinds_control)
target_compile_options(blinds_motion PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(motion_executor_tests tests/motion_executor_tests.cpp)
target_link_libraries(motion_executor_tests PRIVATE blinds_motion)
target_compile_options(motion_executor_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME motion_executor_tests COMMAND motion_executor_tests)
