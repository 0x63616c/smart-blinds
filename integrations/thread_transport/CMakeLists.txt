cmake_minimum_required(VERSION 3.20)
project(smart_blinds_thread_transport LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(LIBCOAP_SOURCE_DIR "" CACHE PATH "Local checkout of pinned libcoap v4.3.5b")
if(NOT EXISTS "${LIBCOAP_SOURCE_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Set LIBCOAP_SOURCE_DIR to a verified libcoap checkout; see README")
endif()
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${LIBCOAP_SOURCE_DIR}" OUTPUT_VARIABLE COAP_REV OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT COAP_REV STREQUAL "851533c3cf63d16984d370ce39d586ecb3694971")
  message(FATAL_ERROR "Unexpected libcoap revision: re-review before changing the pin")
endif()
execute_process(COMMAND git diff --quiet HEAD -- WORKING_DIRECTORY "${LIBCOAP_SOURCE_DIR}" RESULT_VARIABLE COAP_DIRTY)
if(NOT COAP_DIRTY EQUAL 0)
  message(FATAL_ERROR "Pinned libcoap source has tracked changes")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ENABLE_DTLS ON CACHE BOOL "" FORCE)
set(DTLS_BACKEND openssl CACHE STRING "" FORCE)
foreach(feature EXAMPLES DOCS TESTS TCP WS OSCORE PROXY_CODE Q_BLOCK AF_UNIX)
  set(ENABLE_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(ENABLE_THREAD_SAFE ON CACHE BOOL "" FORCE)
find_package(OpenSSL 3.0...<4.0 REQUIRED)
add_subdirectory("${LIBCOAP_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/libcoap")
option(SB_ALLOW_CCM8_POLICY "Explicitly permit the OpenThread 64-bit-tag CCM8 suite; see security notes" OFF)
if(NOT SB_ALLOW_CCM8_POLICY)
  message(FATAL_ERROR "CCM8 requires an explicit scoped security-policy decision; read README then set SB_ALLOW_CCM8_POLICY=ON for the loopback evaluation")
endif()
# No system crypto configuration changes. The only allowed cipher is CCM8;
# OpenSSL's normal security level rejects its 64-bit authentication tag.
target_compile_definitions(coap-3 PRIVATE COAP_OPENSSL_PSK_CIPHERS="PSK-AES128-CCM8:@SECLEVEL=0" COAP_OPENSSL_CIPHERS="PSK-AES128-CCM8:@SECLEVEL=0")
find_package(Threads REQUIRED)
add_library(sb_thread SHARED native/client.cpp)
target_link_libraries(sb_thread PRIVATE coap-3 OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
target_compile_options(sb_thread PRIVATE -Wall -Wextra -Wpedantic -Werror)
option(SB_BUILD_TEST_PEER "Build loopback-only peer fixture (never deploy this)" OFF)
if(SB_BUILD_TEST_PEER)
  target_sources(sb_thread PRIVATE native/test_peer.cpp)
  target_include_directories(sb_thread PRIVATE ../../firmware/include)
endif()
