cmake_minimum_required(VERSION 3.20)
project(smart_blinds_wifi_fixture LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(LIBCOAP_SOURCE_DIR "" CACHE PATH "Pinned vendor libcoap submodule")
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${LIBCOAP_SOURCE_DIR}"
  OUTPUT_VARIABLE REV OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT REV STREQUAL "d9b4031ee61df1a40ecec46fe3817a1f985b3919")
  message(FATAL_ERROR "Wrong Wi-Fi fixture libcoap revision")
endif()
execute_process(COMMAND git diff --quiet HEAD -- WORKING_DIRECTORY "${LIBCOAP_SOURCE_DIR}" RESULT_VARIABLE DIRTY)
if(NOT DIRTY EQUAL 0)
  message(FATAL_ERROR "Unreviewed dependency changes")
endif()
option(SB_ALLOW_CCM8_POLICY "Explicit loopback-only CCM8 evaluation policy" OFF)
if(NOT SB_ALLOW_CCM8_POLICY)
  message(FATAL_ERROR "Read README and explicitly enable the scoped CCM8 evaluation policy")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ENABLE_DTLS ON CACHE BOOL "" FORCE)
set(DTLS_BACKEND openssl CACHE STRING "" FORCE)
foreach(feature EXAMPLES DOCS TESTS TCP WS OSCORE PROXY_CODE Q_BLOCK AF_UNIX)
  set(ENABLE_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(ENABLE_THREAD_SAFE ON CACHE BOOL "" FORCE)
find_package(OpenSSL 3.0...<4.0 REQUIRED)
find_package(Threads REQUIRED)
add_subdirectory("${LIBCOAP_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/libcoap")
# Only this test library's SSL contexts; no system-wide security changes.
target_compile_definitions(coap-3 PRIVATE COAP_OPENSSL_PSK_CIPHERS="PSK-AES128-CCM8:@SECLEVEL=0" COAP_OPENSSL_CIPHERS="PSK-AES128-CCM8:@SECLEVEL=0")
add_library(sb_wifi SHARED ../thread_transport/native/client.cpp native/test_peer.cpp)
target_include_directories(sb_wifi PRIVATE ../../firmware/include ../thread_transport/native)
target_link_libraries(sb_wifi PRIVATE coap-3 OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
target_compile_options(sb_wifi PRIVATE -Wall -Wextra -Wpedantic -Werror)
